**infra/lib/constructs/static-site.ts** -- Adapted from golden paths WebAppConstruct. Per D-01, strip ALL backend resources (no api prop, no apiOrigin, no /api/* behavior). Per D-03, S3 bucket is private with OAC. Per D-07, CloudFront has custom domain aliases and ACM certificate.

```typescript
import { Construct } from "constructs";
import { CfnOutput, DockerImage, Duration, RemovalPolicy } from "aws-cdk-lib";
import * as s3 from "aws-cdk-lib/aws-s3";
import * as cloudfront from "aws-cdk-lib/aws-cloudfront";
import * as origins from "aws-cdk-lib/aws-cloudfront-origins";
import * as s3deploy from "aws-cdk-lib/aws-s3-deployment";
import * as acm from "aws-cdk-lib/aws-certificatemanager";
import { spawnSync } from "child_process";
import * as fs from "fs";
import * as path from "path";

export interface StaticSiteConstructProps {
  frontendAssetPath: string;  // Path to repo root (contains package.json with build script)
  certificate: acm.ICertificate;  // ACM cert for CloudFront
  domainNames: string[];  // e.g., ["shouldwetestthat.com", "www.shouldwetestthat.com"]
}

export class StaticSiteConstruct extends Construct {
  public readonly distribution: cloudfront.Distribution;
  public readonly bucket: s3.Bucket;

  constructor(scope: Construct, id: string, props: StaticSiteConstructProps) {
    super(scope, id);

    // S3 bucket: private access only, no public access (per D-03)
    this.bucket = new s3.Bucket(this, "FrontendBucket", {
      versioned: true,
      blockPublicAccess: s3.BlockPublicAccess.BLOCK_ALL,
      removalPolicy: RemovalPolicy.RETAIN,
      enforceSSL: true,
    });

    // Origin Access Control for secure S3 access from CloudFront
    const oac = new cloudfront.S3OriginAccessControl(this, "OAC", {
      signing: cloudfront.Signing.SIGV4_NO_OVERRIDE,
    });

    const s3Origin = origins.S3BucketOrigin.withOriginAccessControl(
      this.bucket,
      { originAccessControl: oac }
    );

    // CloudFront distribution with custom domain and TLS (per D-07)
    this.distribution = new cloudfront.Distribution(this, "Distribution", {
      defaultBehavior: {
        origin: s3Origin,
        viewerProtocolPolicy: cloudfront.ViewerProtocolPolicy.REDIRECT_TO_HTTPS,
        cachePolicy: cloudfront.CachePolicy.CACHING_OPTIMIZED,
      },
      domainNames: props.domainNames,
      certificate: props.certificate,
      defaultRootObject: "index.html",
      minimumProtocolVersion: cloudfront.SecurityPolicyProtocol.TLS_V1_2_2021,
      // SPA routing: 403/404 -> /index.html (per D-03)
      errorResponses: [
        {
          httpStatus: 403,
          responseHttpStatus: 200,
          responsePagePath: "/index.html",
          ttl: Duration.seconds(0),
        },
        {
          httpStatus: 404,
          responseHttpStatus: 200,
          responsePagePath: "/index.html",
          ttl: Duration.seconds(0),
        },
      ],
    });

    // Deploy built frontend assets to S3 (local bundling preferred, Docker fallback)
    new s3deploy.BucketDeployment(this, "DeployFrontend", {
      sources: [
        s3deploy.Source.asset(props.frontendAssetPath, {
          bundling: {
            image: DockerImage.fromRegistry("node:22-alpine"),
            command: [
              "sh", "-c",
              "npm ci && npm run build && cp -r /asset-input/dist/. /asset-output/",
            ],
            local: {
              tryBundle(outputDir: string, _options: import("aws-cdk-lib").BundlingOptions): boolean {
                const buildResult = spawnSync("npm", ["run", "build"], {
                  cwd: props.frontendAssetPath,
                  stdio: "inherit",
                  env: { ...process.env },
                });
                if (buildResult.status !== 0) return false;
                const distDir = path.join(props.frontendAssetPath, "dist");
                if (!fs.existsSync(distDir)) return false;
                fs.cpSync(distDir, outputDir, { recursive: true });
                return true;
              },
            },
          },
        }),
      ],
      destinationBucket: this.bucket,
      distribution: this.distribution,
      distributionPaths: ["/*"],
    });

    new CfnOutput(this, "CloudFrontURL", {
      value: `https://${this.distribution.distributionDomainName}`,
      description: "CloudFront distribution URL",
    });
  }
}
```

**infra/lib/should-i-test-that-stack.ts** -- Main stack. Creates Route 53 hosted zone (per D-05), ACM certificate with DNS validation (per D-06), and StaticSiteConstruct. Outputs the Route 53 nameservers for GoDaddy delegation.

```typescript
import * as cdk from "aws-cdk-lib/core";
import { CfnOutput } from "aws-cdk-lib";
import { Construct } from "constructs";
import * as route53 from "aws-cdk-lib/aws-route53";
import * as targets from "aws-cdk-lib/aws-route53-targets";
import * as acm from "aws-cdk-lib/aws-certificatemanager";
import { StaticSiteConstruct } from "./constructs/static-site";
import * as path from "path";

const DOMAIN_NAME = "shouldwetestthat.com";

export class ShouldITestThatStack extends cdk.Stack {
  constructor(scope: Construct, id: string, props?: cdk.StackProps) {
    super(scope, id, props);

    // Route 53 hosted zone for DNS management (per D-05)
    const hostedZone = new route53.PublicHostedZone(this, "HostedZone", {
      zoneName: DOMAIN_NAME,
    });

    // ACM certificate for apex + www (per D-06)
    // Stack must be in us-east-1 for CloudFront certificate
    const certificate = new acm.Certificate(this, "SiteCertificate", {
      domainName: DOMAIN_NAME,
      subjectAlternativeNames: [`www.${DOMAIN_NAME}`],
      validation: acm.CertificateValidation.fromDns(hostedZone),
    });

    // Static site: S3 + CloudFront (per D-01, D-03, D-07)
    const site = new StaticSiteConstruct(this, "StaticSite", {
      frontendAssetPath: path.join(__dirname, "../.."),
      certificate,
      domainNames: [DOMAIN_NAME, `www.${DOMAIN_NAME}`],
    });

    // Route 53 A record: apex -> CloudFront
    new route53.ARecord(this, "ApexAlias", {
      zone: hostedZone,
      target: route53.RecordTarget.fromAlias(
        new targets.CloudFrontTarget(site.distribution)
      ),
    });

    // Route 53 AAAA record: apex -> CloudFront (IPv6)
    new route53.AaaaRecord(this, "ApexAliasIPv6", {
      zone: hostedZone,
      target: route53.RecordTarget.fromAlias(
        new targets.CloudFrontTarget(site.distribution)
      ),
    });

    // Route 53 A record: www -> CloudFront
    new route53.ARecord(this, "WwwAlias", {
      zone: hostedZone,
      recordName: "www",
      target: route53.RecordTarget.fromAlias(
        new targets.CloudFrontTarget(site.distribution)
      ),
    });

    // Route 53 AAAA record: www -> CloudFront (IPv6)
    new route53.AaaaRecord(this, "WwwAliasIPv6", {
      zone: hostedZone,
      recordName: "www",
      target: route53.RecordTarget.fromAlias(
        new targets.CloudFrontTarget(site.distribution)
      ),
    });

    // Output nameservers for GoDaddy delegation (per D-05)
    new CfnOutput(this, "NameServers", {
      value: cdk.Fn.join(", ", hostedZone.hostedZoneNameServers!),
      description: "Set these as nameservers in GoDaddy DNS settings",
    });

    new CfnOutput(this, "SiteURL", {
      value: `https://${DOMAIN_NAME}`,
      description: "Production site URL",
    });
  }
}
```
