Create `.github/workflows/deploy.yml` adapted from the golden paths deploy.yml.

Key differences from golden paths:
- No `workflow_dispatch` environment input (single env: prod) -- keep workflow_dispatch for manual redeploy but without environment choice
- No `DD_API_KEY` or `DD_SITE` env vars (per D-12, no backend instrumentation)
- No Lambda tests (`npm test` at root) -- app tests are in the root, run with `npx vitest run`
- No separate `frontend` working-directory step -- this IS the frontend repo (root-level npm ci + build)
- App build happens via BucketDeployment's local bundling (CDK runs `npm run build` at frontendAssetPath), but we also build first to validate
- Use `npx vitest run` (not `npm test` which starts watch mode)

```yaml
name: Deploy

on:
  push:
    branches: [main]
  workflow_dispatch: {}

permissions:
  id-token: write
  contents: read

jobs:
  deploy:
    name: Deploy to AWS
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: "22"
          cache: "npm"

      - name: Configure AWS credentials via OIDC
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE_ARN }}
          aws-region: ${{ secrets.AWS_REGION }}

      - name: Install app dependencies
        run: npm ci

      - name: Run app tests
        run: npx vitest run

      - name: Install infra dependencies
        working-directory: infra
        run: npm ci

      - name: Build CDK
        working-directory: infra
        run: npm run build

      - name: Deploy
        working-directory: infra
        run: npx cdk deploy --require-approval never --outputs-file cdk-outputs.json

      - name: Output deployment URL
        working-directory: infra
        run: |
          CLOUDFRONT_URL=$(cat cdk-outputs.json | python3 -c "import sys,json; data=json.load(sys.stdin); print(next((v for stack in data.values() for k,v in stack.items() if 'CloudFront' in k or 'SiteURL' in k), 'Not found'))")
          echo "## Deployment Complete" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "**App URL**: $CLOUDFRONT_URL" >> $GITHUB_STEP_SUMMARY
          echo "**Commit**: ${{ github.sha }}" >> $GITHUB_STEP_SUMMARY
```

Per D-08: OIDC only (no static keys). Per D-09: push to main triggers. Per D-10: AWS_ROLE_ARN and AWS_REGION secrets.

```yaml
name: CI

on:
  pull_request:
    branches: [main]

jobs:
  test:
    name: Test & Validate
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: "22"
          cache: "npm"

      - name: Install app dependencies
        run: npm ci

      - name: Lint
        run: npm run lint

      - name: Run app tests
        run: npx vitest run

      - name: Build app
        run: npm run build

      - name: Install infra dependencies
        working-directory: infra
        run: npm ci

      - name: Build CDK (TypeScript compile)
        working-directory: infra
        run: npm run build

      - name: CDK Synth (validate CloudFormation template)
        working-directory: infra
        run: npx cdk synth --quiet
        env:
          CDK_DEFAULT_ACCOUNT: "000000000000"
          CDK_DEFAULT_REGION: "us-east-1"
```

Note: CDK synth needs CDK_DEFAULT_ACCOUNT set to a dummy value since our bin file uses `process.env.CDK_DEFAULT_ACCOUNT`. This is fine for template validation in CI.
