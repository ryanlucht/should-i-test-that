         ```typescript
         {
           name: 'wizard-storage',
           storage: createJSONStorage(() => sessionStorage),
           partialize: (state) => ({ inputs: state.inputs }),
           // Sanitize persisted state: remove obsolete keys that would cause type errors
           // This handles the conversionLatencyDays removal (ENG-07) gracefully
           merge: (persistedState: unknown, currentState: WizardStore) => {
             const persisted = persistedState as Partial<WizardStore> | undefined;
             if (persisted?.inputs) {
               // Strip unknown keys from persisted inputs by only keeping known keys
               const knownInputs = { ...currentState.inputs };
               for (const key of Object.keys(persisted.inputs) as (keyof WizardInputs)[]) {
                 if (key in knownInputs) {
                   (knownInputs as Record<string, unknown>)[key] = (persisted.inputs as Record<string, unknown>)[key];
                 }
               }
               return { ...currentState, inputs: knownInputs };
             }
             return currentState;
           },
         }
         ```
