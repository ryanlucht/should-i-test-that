       ```typescript
       /**
        * Maps calculator section indices to the WizardInputs fields that must be
        * non-null for that section to be considered "complete" during URL hydration.
        * This prevents blanket section completion when some inputs are missing.
        */
       const SECTION_REQUIRED_FIELDS: Record<number, (keyof WizardInputs)[]> = {
         0: ['baselineConversionRate', 'annualVisitors', 'valuePerConversion'],  // Baseline
         1: ['priorType'],                                                        // Uncertainty
         2: ['thresholdScenario'],                                                // Threshold
         3: ['testDurationDays', 'dailyTraffic'],                                 // Test Design
       };
       ```

    4. Add a `useEffect` that runs once on mount (empty deps):
       ```typescript
       useEffect(() => {
         if (hydrated.current) return;
         const hash = window.location.hash;
         if (!hash.startsWith('#s=')) return;

         const encoded = hash.slice(3); // Remove "#s="
         const decoded = decodeWizardState(encoded);
         if (!decoded) return; // Invalid/malformed URL -- do nothing, show welcome page normally

         hydrated.current = true;

         // Hydrate store with decoded inputs
         const store = useWizardStore.getState();
         for (const [key, value] of Object.entries(decoded)) {
           store.setInput(key as keyof WizardInputs, value as WizardInputs[keyof WizardInputs]);
         }

         // Enable guide for recipients (D-07)
         store.setGuideEnabled(true);

         // Mark sections completed based on hydrated field validity (not blanket)
         // Only mark a section complete if ALL its required fields are non-null
         for (const [sectionStr, requiredFields] of Object.entries(SECTION_REQUIRED_FIELDS)) {
           const section = Number(sectionStr);
           const allPresent = requiredFields.every((field) => decoded[field] !== null);
           if (allPresent) {
             store.markSectionComplete(section);
           }
         }

         // Store shared baseline for modified-field tracking (D-08)
         store.setSharedBaseline({ ...decoded });

         // Skip welcome page -- go directly to calculator
         setCurrentPage('calculator');

         // Clean the URL hash AFTER successful hydration commit
         // This is done last to preserve recoverability during partial failure
         window.history.replaceState(null, '', window.location.pathname);
       }, []);
       ```

    5. Use `useWizardStore.getState()` (Zustand imperative access) inside the effect -- avoids subscribing to store changes and is correct for one-shot initialization.

    6. Import `WizardInputs` type: `import type { WizardInputs } from '@/types/wizard'`.
