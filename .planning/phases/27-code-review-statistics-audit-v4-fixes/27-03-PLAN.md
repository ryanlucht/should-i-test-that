    **CR-3 fix: Replace btoa/atob with TextEncoder/TextDecoder in url-codec.ts:**

    Replace the `toBase64Url` function:
    ```typescript
    function toBase64Url(input: string): string {
      // UTF-8 safe: encode string to bytes first, then to base64
      const bytes = new TextEncoder().encode(input);
      // Convert Uint8Array to binary string for btoa
      let binary = '';
      for (let i = 0; i < bytes.length; i++) {
        binary += String.fromCharCode(bytes[i]);
      }
      return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }
    ```

    Replace the `fromBase64Url` function:
    ```typescript
    function fromBase64Url(input: string): string | null {
      try {
        const standard = input.replace(/-/g, '+').replace(/_/g, '/');
        const paddingNeeded = (4 - (standard.length % 4)) % 4;
        const padded = standard + '='.repeat(paddingNeeded);
        const binary = atob(padded);
        // Convert binary string back to UTF-8 bytes
        const bytes = new Uint8Array(binary.length);
        for (let i = 0; i < binary.length; i++) {
          bytes[i] = binary.charCodeAt(i);
        }
        return new TextDecoder().decode(bytes);
      } catch {
        return null;
      }
    }
    ```
